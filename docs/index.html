🎮 学习内容

pwd - 查看当前目录
ls - 列出目录内容
cd - 切换目录
mkdir - 创建目录
touch - 创建文件
cp - 复制文件
mv - 移动/重命名
rm - 删除文件
cat - 查看文件内容
echo - 输出文本
